=== Individual Module Test Commands ===
# NOTE: tb files are in tb/, compiled outputs go to sim_build/

# Test Boot ROM + Instruction Memory
iverilog -o sim_build/test_bootrom instr_mem.v tb/test_bootrom_tb.v && vvp sim_build/test_bootrom

# Test Data Memory
iverilog -o sim_build/data_mem_test data_mem.v tb/data_mem_tb.v && vvp sim_build/data_mem_test

# Test ALU
iverilog -o sim_build/alu_test alu.v tb/alu_tb.v && vvp sim_build/alu_test

# Test Register File
iverilog -o sim_build/reg_test reg_file.v tb/reg_file_tb.v && vvp sim_build/reg_test

# Test Program Counter
iverilog -o sim_build/pc_test pc.v tb/pc_tb.v && vvp sim_build/pc_test

# Test Branch Comparator
iverilog -o sim_build/branch_test branch_comp.v tb/branch_comp_tb.v && vvp sim_build/branch_test

# Run all module tests
for mod in sim_build/test_bootrom sim_build/data_mem_test sim_build/alu_test sim_build/reg_test sim_build/pc_test sim_build/branch_test; do
    echo "=== Testing $mod ===" && vvp $mod
done

# View waveforms (VCDs are generated in sim_build/)
gtkwave sim_build/test_bootrom.vcd &
gtkwave sim_build/data_mem_tb.vcd &
gtkwave sim_build/alu_tb.vcd &
gtkwave sim_build/reg_file_tb.vcd &
gtkwave sim_build/pc_tb.vcd &
gtkwave sim_build/branch_comp_tb.vcd &
